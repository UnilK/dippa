cmake_minimum_required(VERSION 4.3)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Option A: Fetch pybind11 automatically (no system dependency needed)
include(FetchContent)
FetchContent_Declare(
  pybind11
  GIT_REPOSITORY https://github.com/pybind/pybind11.git
  GIT_TAG v3.0.4
)
FetchContent_MakeAvailable(pybind11)

if (MSVC)
    # dunno
else()
    add_compile_options(-O3)
endif()

# If you prefer to use a preinstalled pybind11, comment the FetchContent block
# and uncomment the next line:
# find_package(pybind11 CONFIG REQUIRED)

# Build the Python extension module
project(propagationmod LANGUAGES CXX)
pybind11_add_module(propagationmod src/propagationmod.cpp)

project(resample LANGUAGES CXX)
pybind11_add_module(resamplemod src/resamplemod.cpp)
